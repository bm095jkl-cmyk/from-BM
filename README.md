local function createSalapaoExtra(ctx, descriptors)
    local X={defs={},values={},layers={},notes={},timers={}}
    local live=true
    for _,d in ipairs(descriptors) do X.defs[d.id]=d;X.values[d.id]=d.default end
    for i=1,5 do X.notes[i]={title='สมุด '..i,body='',undo=''} end
    for i=1,6 do X.timers[i]={duration=300,label='เวลา '..i,remaining=300,running=false,repeatOn=false} end
    local function errorText(s) return false,s end
    local function color(raw)
        local r,g,b=tostring(raw):match('^%s*(%d+)%s*,%s*(%d+)%s*,%s*(%d+)%s*$')
        if not r or tonumber(r)>255 or tonumber(g)>255 or tonumber(b)>255 then return nil end
        return ctx.rgb(tonumber(r),tonumber(g),tonumber(b))
    end
    local function remaining(t,now) return t.running and math.max(0,t.deadline-now) or t.remaining end
    local function layer(i)
        local state={}
        for _,d in ipairs(descriptors)do if d.group=='layer'and d.slot==i then state[d.field]=X.values[d.id]end end
        local function global(k)return X.values['salapao_workspace_1_'..k]end
        if not state.enabled or global('enabled')==false then if X.layers[i]then X.layers[i]:Destroy();X.layers[i]=nil end;return end
        state.x=math.max(0,math.min(100,state.x+(global('xOffset')or 0)))
        state.y=math.max(0,math.min(100,state.y+(global('yOffset')or 0)))
        state.fontSize=state.fontSize*(global('textScale')or 1)
        state.rotation=state.rotation+(global('rotation')or 0)
        state.opacity=state.opacity*(global('opacity')or 1)
        state.bgOpacity=state.bgOpacity*(global('opacity')or 1)
        local obj=X.layers[i]
        if not obj then obj=ctx.label(i);X.layers[i]=obj end
        ctx.paint(obj,state)
    end
    local function noteSet(n,body) n.undo=n.body;n.body=body end
    local function snapshot()
        local settings={};for _,d in ipairs(descriptors)do if d.group=='layer'and d.kind~='action'then settings[d.id]=X.values[d.id]end end
        return{schema=1,settings=settings}
    end
    local function import(doc)
        if type(doc)~='table'or doc.schema~=1 or type(doc.settings)~='table'then return false,'รูปแบบสตูดิโอไม่ถูกต้อง'end
        for id,v in pairs(doc.settings)do
            local d=X.defs[id];if not d or d.group~='layer'or d.kind=='action'then return false,'พบการตั้งค่าที่ไม่รองรับ'end
            if d.kind=='number'and(type(v)~='number'or v~=v or v<d.min or v>d.max)then return false,'ค่าตัวเลขไม่ถูกต้อง'end
            if d.kind=='toggle'and type(v)~='boolean'then return false,'ค่าเปิดปิดไม่ถูกต้อง'end
            if d.kind=='text'and(type(v)~='string'or #v>32000 or d.color and not color(v))then return false,'ข้อความหรือสีไม่ถูกต้อง'end
            if d.kind=='choice'then local ok=false;for _,s in ipairs(d.choices)do if v==s then ok=true end end;if not ok then return false,'ตัวเลือกไม่ถูกต้อง'end end
        end
        for id,v in pairs(doc.settings)do X.values[id]=v end
        for i=1,24 do layer(i)end
        return true
    end
    function X.get(id)
        local d=X.defs[id];if not d then return nil end
        if d.group=='note'and(d.field=='body'or d.field=='title')then return X.notes[d.slot][d.field]end
        return X.values[id]
    end
    function X.set(id,v)
        if not live then return errorText('โปรปิดแล้ว')end
        local d=X.defs[id];if not d then return errorText('ไม่พบคำสั่ง')end
        if d.kind=='number'then
            v=tonumber(v);if not v or v~=v or v==math.huge or v==-math.huge then return errorText('กรอกตัวเลขที่ถูกต้อง')end
            v=math.max(d.min,math.min(d.max,v))
        elseif d.kind=='toggle'then if type(v)~='boolean'then return errorText('ต้องเป็นเปิดหรือปิด')end
        elseif d.kind=='choice'then
            local found=false;for _,c in ipairs(d.choices)do if c==v then found=true end end
            if not found then return errorText('ตัวเลือกไม่ถูกต้อง')end
        elseif d.kind=='text'then
            v=tostring(v or'');if #v>32000 then return errorText('ข้อความยาวเกิน 32,000 ไบต์')end
            if d.color and not color(v)then return errorText('ใช้สี R,G,B ช่วง 0–255')end
        end
        if d.group=='layer'then
            if d.field=='reset'then
                for _,spec in ipairs(descriptors)do if spec.group=='layer'and spec.slot==d.slot then X.values[spec.id]=spec.default end end
            else X.values[id]=v end
            layer(d.slot)
        elseif d.group=='note'then
            local n=X.notes[d.slot];local field=d.field
            if field=='title'then n.title=v
            elseif field=='body'then noteSet(n,v)
            elseif field=='append'then noteSet(n,n.body..(n.body==''and''or'\n')..v)
            elseif field=='replace'then
                local find,replacement=v:match('^(.-)%s*=>%s*(.*)$')
                if not find or find==''then return errorText('รูปแบบ: ข้อความเดิม => ข้อความใหม่')end
                local out,start={},1
                while true do local a,b=n.body:find(find,start,true);if not a then out[#out+1]=n.body:sub(start);break end;out[#out+1]=n.body:sub(start,a-1);out[#out+1]=replacement;start=b+1 end
                noteSet(n,table.concat(out))
            elseif field=='copy'then local ok,err=ctx.copy(n.title..'\n\n'..n.body);if not ok then return errorText(err)end
            elseif field=='clear'then noteSet(n,'')
            elseif field=='undo'then n.body,n.undo=n.undo,n.body
            elseif field=='stats'then
                local words=0;for _ in n.body:gmatch('%S+')do words=words+1 end
                local chars=ctx.length(n.body);ctx.report(d,string.format('%s\n%d ตัวอักษร · %d กลุ่มคำ · %d ไบต์',n.title,chars,words,#n.body))
            elseif field=='save'then local ok,err=ctx.save(d.slot,n);if not ok then return errorText(err)end
            elseif field=='load'then
                local doc,err=ctx.load(d.slot)
                if type(doc)~='table'or type(doc.title)~='string'or type(doc.body)~='string'or #doc.body>32000 then return errorText(err or'ไฟล์สมุดไม่ถูกต้อง')end
                noteSet(n,doc.body);n.title=doc.title
            end
            if d.kind=='text'then X.values[id]=v end
        elseif d.group=='workspace'then
            local field=d.field
            if d.kind=='number'or d.kind=='toggle'then X.values[id]=v
            elseif field=='export'then ctx.report(d,ctx.encode(snapshot()))
            elseif field=='copy'then local ok,e=ctx.copy(ctx.encode(snapshot()));if not ok then return false,e end
            elseif field=='import'then local doc,e=ctx.decode(v);if not doc then return false,e end;local ok,why=import(doc);if not ok then return false,why end
            elseif field=='save'then local ok,e=ctx.saveWorkspace(snapshot());if not ok then return false,e end
            elseif field=='load'then local doc,e=ctx.loadWorkspace();if not doc then return false,e end;local ok,why=import(doc);if not ok then return false,why end
            elseif field=='summary'then
                local enabled=0;for i=1,24 do if X.values['salapao_layer_'..i..'_enabled']then enabled=enabled+1 end end
                ctx.report(d,tostring(enabled)..' / 24 เลเยอร์เปิดอยู่')
            elseif field=='hideAll'or field=='showAll'then for i=1,24 do X.values['salapao_layer_'..i..'_enabled']=field=='showAll'end
            elseif field=='resetAll'then for _,spec in ipairs(descriptors)do if spec.group=='layer'then X.values[spec.id]=spec.default end end
            elseif field=='centerAll'then for i=1,24 do X.values['salapao_layer_'..i..'_x']=50;X.values['salapao_layer_'..i..'_y']=50 end
            elseif field=='centerSelected'then local i=math.floor(X.values.salapao_workspace_1_selected);X.values['salapao_layer_'..i..'_x']=50;X.values['salapao_layer_'..i..'_y']=50 end
            for i=1,24 do layer(i)end
            ctx.grid(X.values.salapao_workspace_1_grid,X.values.salapao_workspace_1_gridSize)
        elseif d.group=='timer'then
            local t=X.timers[d.slot];local now=ctx.now()
            if d.field=='duration'then t.duration=v;if not t.running then t.remaining=v end;X.values[id]=v
            elseif d.field=='label'then t.label=v;X.values[id]=v
            elseif d.field=='repeatOn'then t.repeatOn=v;X.values[id]=v
            elseif d.field=='start'then if not t.running then if t.remaining<=0 then t.remaining=t.duration end;t.deadline=now+t.remaining;t.running=true end
            elseif d.field=='pause'then t.remaining=remaining(t,now);t.running=false
            elseif d.field=='reset'then t.running=false;t.remaining=t.duration end
        end
        return true
    end
    function X.tick(now)
        if not live then return end
        for _,t in ipairs(X.timers)do
            if t.running and now>=t.deadline then
                ctx.notify('ครบเวลา · '..t.label)
                if t.repeatOn then t.deadline=now+t.duration-(now-t.deadline)%t.duration;t.remaining=t.duration
                else t.running=false;t.remaining=0 end
            end
        end
    end
    function X.summary()
        local lines={};local now=ctx.now()
        for _,t in ipairs(X.timers)do local r=remaining(t,now);lines[#lines+1]=string.format('%s  %02d:%02d  %s',t.label,math.floor(r/60),math.floor(r%60),t.running and'กำลังนับ'or'หยุด')end
        return table.concat(lines,'\n')
    end
    function X.destroy()
        live=false;for _,obj in pairs(X.layers)do obj:Destroy()end;X.layers={}
        for _,t in ipairs(X.timers)do t.running=false end
    end
    return X
end
